# 스냅샷 대신 DOM에서 직접 꺼낸다 >>> page.evaluate("document.body.innerText") 신착 도서 제목 저자 상태 습관의 힘 찰스 두히그 대출 가능 클린 코드 로버트 마틴 대출 중 도서관 이용 안내 상세 보기 >>> page.evaluate("[...document.querySelectorAll('tr[data-book-id]')] .map(r => r.dataset.bookId + ' ' + r.dataset.isbn)") ['1001 9780000000001', '1002 9780000000002'] >>> len(page.content()) 719 # HTML 원본 전체 (글자 수)
