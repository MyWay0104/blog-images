litellm_demo.py — LiteLLM Python SDK (공식 문서의 예제를 바탕으로 재구성)
from litellm import completion # 제공사가 달라도 같은 함수, 같은 OpenAI 형식 메시지 messages = [{"role": "user", "content": "이 독서 노트를 세 줄로 요약해 줘: ..."}] r1 = completion(model="openai/gpt-5.4-nano", messages=messages) r2 = completion(model="anthropic/<모델명>", messages=messages) # 응답 형태도 제공사와 무관하게 OpenAI 형식으로 통일 print(r1.choices[0].message.content) print(r2.choices[0].message.content) # 모델 이름 규칙: "제공사/모델" (슬래시). 모델명은 각 제공사 문서의 이름을 그대로 넣는다
