gateway_client.py — 게이트웨이를 거쳐 부르기 (LiteLLM 공식 문서 예제를 바탕으로 재구성)
import openai # 앱은 제공사 키를 모른다. 게이트웨이가 준 "가상 키"와 주소만 안다 client = openai.OpenAI( api_key="sk-your-virtual-key", base_url="http://localhost:4000", # LiteLLM 게이트웨이 ) # 호출 코드는 평범한 OpenAI SDK 그대로. 어느 제공사로 갈지는 게이트웨이 설정이 정한다 r = client.chat.completions.create( model="reading-notes-summary", # 게이트웨이에 등록한 별칭 messages=[{"role": "user", "content": "이 독서 노트를 세 줄로 요약해 줘: ..."}], ) print(r.choices[0].message.content)
