langchain_demo.py — LangChain 표준 모델 인터페이스 (공식 문서 기준 재구성)
from langchain.chat_models import init_chat_model # 제공사:모델 (콜론) — 제공사별 패키지(langchain-openai 등)가 설치되어 있어야 함 m1 = init_chat_model("openai:gpt-5.4-nano", temperature=0) m2 = init_chat_model("anthropic:<모델명>", temperature=0) # 호출 방법은 같다. 돌아오는 것은 LangChain 메시지 객체(AIMessage) out = m1.invoke("이 독서 노트를 세 줄로 요약해 줘: ...") print(out.content) # 모델은 "부품"이고, 그 위에 도구 호출·에이전트·구조화 출력이 올라간다 # agent = create_agent(model=m1, tools=[...])
